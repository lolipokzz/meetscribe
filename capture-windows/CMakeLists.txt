cmake_minimum_required(VERSION 3.24)
project(meetscribe_capture VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(CTest)
if(NOT WIN32)
    if(BUILD_TESTING)
        add_executable(wave-test tests/wave_test.cpp)
        target_compile_features(wave-test PRIVATE cxx_std_20)
        add_test(NAME wave-integrity COMMAND wave-test)
        add_executable(audio-test tests/audio_test.cpp)
        target_compile_features(audio-test PRIVATE cxx_std_20)
        add_test(NAME audio-conversion COMMAND audio-test)
        return()
    endif()
    message(FATAL_ERROR "Capture executable targets Windows 10/11 x64; use BUILD_TESTING for portable tests.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "meetscribe-capture requires an x64 toolchain.")
endif()

if(MSVC AND NOT MSVC_CXX_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "Use the MSVC x64 target architecture, not ARM64.")
endif()

add_executable(meetscribe-capture src/main.cpp)
target_compile_features(meetscribe-capture PRIVATE cxx_std_20)
set_target_properties(meetscribe-capture PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(meetscribe-capture PRIVATE /W4 /permissive-)
else()
    target_compile_options(meetscribe-capture PRIVATE -Wall -Wextra -Wpedantic)
endif()

target_compile_definitions(meetscribe-capture PRIVATE UNICODE _UNICODE)
target_link_libraries(meetscribe-capture PRIVATE ole32 uuid ksuser)
if(MINGW)
    target_link_options(meetscribe-capture PRIVATE -municode -static)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(wave-test tests/wave_test.cpp)
    target_compile_features(wave-test PRIVATE cxx_std_20)
    add_test(NAME wave-integrity COMMAND wave-test)
    add_executable(audio-test tests/audio_test.cpp)
    target_compile_features(audio-test PRIVATE cxx_std_20)
    add_test(NAME audio-conversion COMMAND audio-test)
    add_test(NAME capture-help COMMAND meetscribe-capture --help)
    set_tests_properties(capture-help PROPERTIES PASS_REGULAR_EXPRESSION "--directory")
endif()
install(TARGETS meetscribe-capture RUNTIME DESTINATION bin)
